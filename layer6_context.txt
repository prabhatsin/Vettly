def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(bearer_scheme),
    db: Session = Depends(get_db),
) -> User:
    unauthorized = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,

        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = decode_access_token(credentials.credentials)
        user_id = uuid.UUID(payload["sub"])

    return user

"""Scoring worker. Run from the api folder:  python -m app.worker"""

import asyncio
import logging
import uuid

from app.db.session import get_db
from app.repositories.scoring import ReportRepository
from app.services.scoring_queue import ScoringQueue
from app.services.scoring_runner import ScoringJob
import time
import redis

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s %(message)s")
logger = logging.getLogger("worker")


def process(application_id: str) -> None:
    """Score one application. Any error is stored on the report instead of crashing."""
    app_uuid = uuid.UUID(application_id)
    session_gen = get_db()
    db = next(session_gen)
    try:
        asyncio.run(ScoringJob(db).run(app_uuid))
    except Exception as exc:
        logger.exception("scoring failed for %s", application_id)
        db.rollback()
        ReportRepository(db).save(app_uuid, status="failed", error=str(exc)[:500])
    finally:
        session_gen.close()


def main() -> None:
    queue = ScoringQueue()
    back = queue.requeue_stale()
    if back:
        logger.info("requeued %d unfinished jobs", back)
    logger.info("worker ready")
    try:
        while True:

            try:
                application_id = queue.reserve(timeout=5)
            except (redis.exceptions.TimeoutError, redis.exceptions.ConnectionError):
                logger.warning("redis not reachable, retrying")
                time.sleep(2)
                continue

            if application_id is None:
                continue
            logger.info("scoring %s", application_id)
            process(application_id)
            queue.ack(application_id)
    except KeyboardInterrupt:
        logger.info("worker stopped")


if __name__ == "__main__":
    main()

class NotFound(Exception):
    """Missing, or not owned by this employer. Both look the same from outside."""


class EmployerService:
    def __init__(self, db: Session):
        self.applications = ApplicationRepository(db)
        self.jobs = JobRepository(db)
        self.transcripts = TranscriptRepository(db)
        self.reports = ReportRepository(db)
        self.rubrics = RubricRepository(db)

    def _own_job(self, job_id: uuid.UUID, employer: User) -> Job:
        job = self.jobs.get_by_id(job_id)
        if job is None or job.employer_id != employer.id:
            raise NotFound()
        return job

    def own_application(self, application_id: uuid.UUID, employer: User) -> Application:
        application = self.applications.get_by_id(application_id)
        if application is None:
            raise NotFound()
        self._own_job(application.job_id, employer)
        return application

    def candidates(self, job_id: uuid.UUID, employer: User) -> list[CandidateRow]:
        self._own_job(job_id, employer)
        return [
            CandidateRow(
                application_id=app.id,
                candidate_email=user.email,
                application_status=app.status.value,
                report_status=report.status if report else None,
                overall_score=report.overall_score if report else None,
                recommendation=report.recommendation if report else None,
            )
            for app, user, report in self.reports.list_for_job(job_id)
        ]

    def report(self, application_id: uuid.UUID, employer: User) -> InterviewReport:
        self.own_application(application_id, employer)
        row = self.reports.get_by_application(application_id)
        if row is None:
            raise NotFound()
        return row

    def replay(self, application_id: uuid.UUID, employer: User) -> ReplayOut:
        self.own_application(application_id, employer)
        entries = self.transcripts.list_by_application(application_id)
        turns = merge_turns([{"seq": e.seq, "role": e.role, "text": e.text} for e in entries])
        evidence: list[ReplayEvidence] = []
        row = self.reports.get_by_application(application_id)
        if row is not None and row.report:
            for c in Report.model_validate(row.report).criteria:
                evidence += [
                    ReplayEvidence(
                        criterion_key=c.key, criterion_name=c.name, turn=ev.turn, quote=ev.quote
                    )
                    for ev in c.evidence
                ]
        return ReplayOut(
            turns=[ReplayTurn(index=i, **t) for i, t in enumerate(turns)],
            evidence=evidence,
        )

    def get_rubric(self, job_id: uuid.UUID, employer: User) -> RubricOut:
        self._own_job(job_id, employer)
        row = self.rubrics.get_by_job(job_id)
        if row is None:
            return RubricOut(job_id=job_id, is_default=True, criteria=DEFAULT_RUBRIC)
        return RubricOut(
            job_id=job_id,
            is_default=False,
            criteria=[Criterion.model_validate(c) for c in row.criteria],
        )

    def set_rubric(
        self, job_id: uuid.UUID, employer: User, criteria: list[Criterion]
    ) -> RubricOut:
        self._own_job(job_id, employer)
        self.rubrics.upsert(job_id, [c.model_dump(mode="json") for c in criteria])
        return RubricOut(job_id=job_id, is_default=False, criteria=criteria)

class ApplicationService:
    def __init__(self, db: Session):
        self.db = db
        self.applications = ApplicationRepository(db)
        self.jobs = JobRepository(db)

    def apply(self, candidate: User, job_id: uuid.UUID, resume: UploadFile) -> Application:
        '''
        checks the job exists and is open, blocks duplicates, saves the PDF, and creates the application 
        row for the logged-in candidate.
        '''
        job = self.jobs.get_by_id(job_id)
        if job is None or not job.is_active:
            raise JobNotFound()
        if self.applications.get_by_job_and_candidate(job_id, candidate.id):
            raise AlreadyApplied()

        filename = save_resume(resume)
        text = extract_text(filename)
        try:
            return self.applications.create(job_id, candidate.id, filename, text)
        except IntegrityError:
            self.db.rollback()
            raise AlreadyApplied()

    def list_mine(self, candidate: User) -> list[Application]:
        return self.applications.list_by_candidate(candidate.id)

    def list_for_job(self, job_id: uuid.UUID, employer: User) -> list[Application]:
        job = self.jobs.get_by_id(job_id)
        if job is None:
            raise JobNotFound()
        if job.employer_id != employer.id:
            raise NotJobOwner()
        return self.applications.list_by_job(job_id)


    def get_for_candidate(self, application_id: uuid.UUID, candidate: User) -> Application:
        '''
        loads an application by id and returns it only if it belongs to the logged-in candidate, 
        otherwise it raises ApplicationNotFound, so a candidate can never get a token for someone 
        else's application.
        
        '''
        application = self.applications.get_by_id(application_id)
        if application is None or application.candidate_id != candidate.id:
            raise ApplicationNotFound()
        return application===== api/app/services/scoring_runner.py

class GeminiScorerClient:
    """Adapts the sync Gemini client to the `generate(prompt) -> ScorerOutput` shape."""

    def __init__(self, client=None, model: str | None = None):
        self.client = client or genai.Client(api_key=settings.google_api_key)
        self.model = model or settings.scoring_model or settings.plan_model

    async def __call__(self, prompt: str) -> ScorerOutput:
        return await asyncio.to_thread(self._call, prompt)

    def _call(self, prompt: str) -> ScorerOutput:
        response = self.client.models.generate_content(
            model=self.model,
            contents=prompt,
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                response_schema=ScorerOutput,
                temperature=0.0,  # same transcript, similar scores on a re-run
            ),
        )
        return ScorerOutput.model_validate_json(response.text)


def rubric_for_job(db: Session, job_id: uuid.UUID) -> list[Criterion]:
    row = RubricRepository(db).get_by_job(job_id)
    if row is None:
        return DEFAULT_RUBRIC
    return [Criterion.model_validate(c) for c in row.criteria]


class ScoringJob:
    def __init__(self, db: Session, generate: Generate | None = None):
        self.generate = generate or GeminiScorerClient()
        self.applications = ApplicationRepository(db)
        self.jobs = JobRepository(db)
        self.transcripts = TranscriptRepository(db)
        self.reports = ReportRepository(db)
        self.db = db

    async def run(self, application_id: uuid.UUID) -> None:
        application = self.applications.get_by_id(application_id)
        if application is None:
            logger.warning("scoring: application %s not found", application_id)
            return
        job = self.jobs.get_by_id(application.job_id)
        rubric = rubric_for_job(self.db, application.job_id)
        entries = self.transcripts.list_by_application(application_id)
        turns = merge_turns(
            [{"seq": e.seq, "role": e.role, "text": e.text} for e in entries]
        )

        self.reports.save(application_id, status="pending")
        context = {
            "job_title": job.title if job else "",
            "resume_excerpt": (application.resume_text or "")[:MAX_RESUME_CHARS],
        }
        report = await score_interview(turns, self.generate, rubric, context)
        self.reports.save(
            application_id,
            status="done",
            overall_score=report.overall_score,
            recommendation=report.recommendation,
            report=report.model_dump(mode="json"),
            rubric=[c.model_dump(mode="json") for c in rubric],
        )
        logger.info(
            "scoring: %s done, overall %s, %s",
            application_id, report.overall_score, report.recommendation,
        )

class ApplicationRepository:
    """All database access for applications: create, get, list by candidate or job."""

    def __init__(self, db: Session):
        self.db = db

    def create(self,job_id: uuid.UUID,candidate_id: uuid.UUID,resume_path: str,resume_text: str | None = None,
    ) -> Application:
        application = Application(
            job_id=job_id,
            candidate_id=candidate_id,
            resume_path=resume_path,
            resume_text=resume_text,
        )
        self.db.add(application)
        self.db.commit()
        self.db.refresh(application)
        return application

    def get_by_id(self, application_id: uuid.UUID) -> Application | None:
        return self.db.get(Application, application_id)

    def get_by_job_and_candidate(
        self, job_id: uuid.UUID, candidate_id: uuid.UUID
    ) -> Application | None:
        stmt = select(Application).where(
            Application.job_id == job_id, Application.candidate_id == candidate_id
        )
        return self.db.scalar(stmt)

    def list_by_candidate(self, candidate_id: uuid.UUID) -> list[Application]:
        stmt = (
            select(Application)
            .where(Application.candidate_id == candidate_id)
            .order_by(Application.created_at.desc())
        )
        return list(self.db.scalars(stmt))

        stmt = (
            select(Application, User.email)
            .join(User, User.id == Application.candidate_id)
            .where(Application.job_id == job_id)
            .order_by(Application.created_at.desc())
        )
        return [(a, email) for a, email in self.db.execute(stmt).all()]

class RubricRepository:
    def __init__(self, db: Session):
        self.db = db

    def get_by_job(self, job_id: uuid.UUID) -> Rubric | None:
        return self.db.scalar(select(Rubric).where(Rubric.job_id == job_id))

    def upsert(self, job_id: uuid.UUID, criteria: list[dict]) -> Rubric:
        row = self.get_by_job(job_id)
        if row is None:
            row = Rubric(job_id=job_id, criteria=criteria)
            self.db.add(row)
        else:
            row.criteria = criteria
        self.db.commit()
        return row


class ReportRepository:
    def __init__(self, db: Session):
        self.db = db

    def get_by_application(self, application_id: uuid.UUID) -> InterviewReport | None:
        return self.db.scalar(
            select(InterviewReport).where(InterviewReport.application_id == application_id)
        )

    def save(
        self,
        application_id: uuid.UUID,
        *,
        status: str,
        overall_score: float | None = None,
        recommendation: str | None = None,
        report: dict | None = None,
        rubric: list | None = None,
        error: str | None = None,
    ) -> InterviewReport:
        """Create or replace the single report row for this application."""
        row = self.get_by_application(application_id)
        if row is None:
            row = InterviewReport(application_id=application_id)
            self.db.add(row)
        row.status = status
        row.overall_score = overall_score
        row.recommendation = recommendation
        row.report = report
        row.rubric = rubric
        row.error = error
        self.db.commit()
        return row

    def list_for_job(
        self, job_id: uuid.UUID
    ) -> list[tuple[Application, User, InterviewReport | None]]:
        """Every application for a job with its candidate and report, best score first."""
        stmt = (
            select(Application, User, InterviewReport)
            .join(User, User.id == Application.candidate_id)
            .outerjoin(InterviewReport, InterviewReport.application_id == Application.id)
            .where(Application.job_id == job_id)
            .order_by(InterviewReport.overall_score.desc().nulls_last(), Application.created_at)
        )
        return [tuple(row) for row in self.db.execute(stmt).all()]

def good_generate(quote="I used the watchdog library and an event driven design"):
    seen = []

    async def generate(prompt):
        seen.append(prompt)
        return ScorerOutput(
            score=4, reason="Clear explanation with a concrete mechanism.",
            evidence=[Evidence(turn=1, quote=quote)],
        )

    generate.seen = seen
    return generate

const API_URL = process.env.NEXT_PUBLIC_API_URL;

export const getToken = () =>
  typeof window === "undefined" ? null : localStorage.getItem("token");
export const setToken = (t: string) => localStorage.setItem("token", t);
export const clearToken = () => localStorage.removeItem("token");

export async function api<T>(path: string, options: RequestInit = {}): Promise<T> {
  const headers = new Headers(options.headers);
  const token = getToken();
  if (token) headers.set("Authorization", `Bearer ${token}`);
  if (options.body && !(options.body instanceof FormData)) {
    headers.set("Content-Type", "application/json");
  }

  const res = await fetch(`${API_URL}${path}`, { ...options, headers });
  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    const detail = typeof body.detail === "string" ? body.detail : "Request failed";
    throw new Error(detail);
  }
  return res.json();

export async function login(email: string, password: string): Promise<User> {
  const res = await api<{ access_token: string }>("/auth/login", {
    method: "POST",
    body: JSON.stringify({ email, password }),
  });
  setToken(res.access_token);
  return api<User>("/auth/me");
}

export const homeFor = (role: Role) => (role === "employer" ? "/employer" : "/jobs");

export function useUser(requiredRole?: Role) {
  const router = useRouter();
  const [user, setUser] = useState<User | null>(null);

  useEffect(() => {
    if (!getToken()) {
      router.replace("/login");
      return;
    }
    api<User>("/auth/me")
      .then((u) => {
        if (requiredRole && u.role !== requiredRole) router.replace(homeFor(u.role));
        else setUser(u);
      })
      .catch(() => {
        clearToken();
        router.replace("/login");
      });
  }, [router, requiredRole]);

  return user;

  const [form, setForm] = useState({
    title: "", description: "", location: "", employment_type: "full_time",
  });
  const [error, setError] = useState("");

  const load = useCallback(
    () => api<Job[]>("/jobs/mine").then(setJobs).catch((e) => setError(e.message)),
    []
  );
  useEffect(() => {
    if (user) load();
  }, [user, load]);

  async function create(e: React.FormEvent) {
    e.preventDefault();
    setError("");
    try {
      await api("/jobs", {
        method: "POST",
        body: JSON.stringify({ ...form, location: form.location || null }),
      });
      setForm({ title: "", description: "", location: "", employment_type: "full_time" });
      load();
    } catch (err) {
      setError((err as Error).message);
    }
  }

  async function toggle(job: Job) {

    load();
  }

  async function showApps(jobId: string) {
    const list = await api<CandidateRow[]>(`/employer/jobs/${jobId}/candidates`);
    setApps((prev) => ({ ...prev, [jobId]: list }));
  }

          <textarea className="rounded border p-2" placeholder="Description (min 10 chars)" required minLength={10}
            value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} />
          <input className="rounded border p-2" placeholder="Location (optional)"
            value={form.location} onChange={(e) => setForm({ ...form, location: e.target.value })} />
          <select className="rounded border p-2" value={form.employment_type}
            onChange={(e) => setForm({ ...form, employment_type: e.target.value })}>
            <option value="full_time">Full time</option>
            <option value="part_time">Part time</option>
            <option value="contract">Contract</option>
          </select>
          {error && <p className="text-sm text-red-600">{error}</p>}
